# プロジェクト セットアップ

## 初期セットアップ

`eslint.config.mjs`:

```js
import yasainetConfig from "@yasainet/eslint-config";
import { defineConfig, globalIgnores } from "eslint/config";
import nextVitals from "eslint-config-next/core-web-vitals";
import nextTs from "eslint-config-next/typescript";

const eslintConfig = defineConfig([
  ...nextVitals,
  ...nextTs,
  // Override default ignores of eslint-config-next.
  globalIgnores([
    // Default ignores of eslint-config-next:
    ".next/**",
    "out/**",
    "build/**",
    "next-env.d.ts",
  ]),
  ...yasainetConfig,
]);

export default eslintConfig;
```

## monorepo

package ごとに `eslint.config.mjs` を置き、その package を cwd にして eslint を実行する (`npm run lint --workspaces`)。rule の glob (`src/**`) は cwd 相対のため。

`createConfig(options)` で package に合わせる:

```js
import { createConfig } from "@yasainet/eslint-config";

export default [
  ...createConfig({
    // import の root として扱う prefix。workspace package を @/ と同じ層の rule で見る
    aliases: ["@/", "@anglers/core/"],
    // feature 名の元になる Supabase の型 file。cwd 相対。別 package にあるときに指定する
    typesPath: "../../packages/core/src/lib/supabase/types.ts",
    // globals.css が無い package では tailwind の rule を外す
    tailwindcss: false,
  }),
];
```

default export は `createConfig()` と同じ (alias は `@/` だけ、型 file は `src/lib/supabase/types.ts`、tailwind あり)。

`tsconfig.json`:

```diff
 {
   "compilerOptions": {
     // ...
     "paths": {
-      "@/*": ["./*"]
+      "@/*": ["./src/*"]
     }
   }
 }
```

## shadcn/ui

```sh
mkdir -p src/utils/
mv src/lib/utils.ts src/utils/cn.ts
```

`components.json`:

```diff
 {
   // ...
   "aliases": {
     "components": "@/components",
-    "utils": "@/lib/utils",
+    "utils": "@/utils/cn",
     "ui": "@/components/ui",
     "lib": "@/lib",
     "hooks": "@/hooks"
   },
 }
```
